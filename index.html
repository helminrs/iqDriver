Set Default Lokasi Jemput Presisi Sesuai GPS HP saat Aplikasi Dibuka function centerToMyLocation() { if (navigator.geolocation) { navigator.geolocation.getCurrentPosition((pos) => { userCurrentLat = pos.coords.latitude; userCurrentLng = pos.coords.longitude; const currentLatLng = L.latLng(userCurrentLat, userCurrentLng); map.setView(currentLatLng, 16); // Set otomatis Titik JEMPUT ke posisi presisi GPS HP saat ini pickupCoords = currentLatLng; const coordStr = `${userCurrentLat.toFixed(5)}, ${userCurrentLng.toFixed(5)}`; document.getElementById('pickupLocation').value = coordStr; if (pickupMarker) map.removeLayer(pickupMarker); pickupMarker = L.marker(currentLatLng, {icon: redPinIcon}).addTo(map); // Jika titik tujuan belum di-set, buat default tujuan di sekitaran lokasi GPS (misal +0.005 derajat offset) if (!dropoffCoords) { const defaultDropoffLat = userCurrentLat + 0.005; const defaultDropoffLng = userCurrentLng + 0.005; dropoffCoords = L.latLng(defaultDropoffLat, defaultDropoffLng); document.getElementById('dropoffLocation').value = `${defaultDropoffLat.toFixed(5)}, ${defaultDropoffLng.toFixed(5)}`; if (dropoffMarker) map.removeLayer(dropoffMarker); dropoffMarker = L.marker(dropoffCoords, {icon: finishFlagIcon}).addTo(map); } // Hitung ulang argo secara otomatis calculateFare(); checkLocationButtonState(); }, (err) => { alert("Gagal mengambil titik GPS presisi. Pastikan akses lokasi di-izinkan."); }, { enableHighAccuracy: true // Memastikan akurasi tinggi GPS HP }); } } // 2. Logika Perhitungan Argo / Ongkos Presisi Sesuai Rentang Meter function calculateFare() { if (pickupCoords && dropoffCoords) { const distanceInMeters = map.distance(pickupCoords, dropoffCoords); // Jarak dalam meter let fare = 0; if (distanceInMeters <= 3600) { // Minimum Rp 9.000 untuk jarak 1m - 3.6km (3600m) fare = 9000; } else { // Jarak di atas 3,6 km (3600 meter) const extraMeters = distanceInMeters - 3600; // Minimum dasar Rp 9.000 fare = 9000; // Hitung km utuh di atas 3,6 km const fullKm = Math.floor(extraMeters / 1000); fare += fullKm * 2500; // Hitung sisa meter (modulus 1000m) berdasarkan aturan pecahan const remainderMeters = Math.round(extraMeters % 1000); if (remainderMeters >= 1 && remainderMeters <= 199) { fare += 500; } else if (remainderMeters >= 200 && remainderMeters <= 399) { fare += 1000; } else if (remainderMeters >= 400 && remainderMeters <= 599) { fare += 1500; } else if (remainderMeters >= 600 && remainderMeters <= 799) { fare += 2000; } else if (remainderMeters >= 800) { fare += 2500; } } computedFare = fare; // Format Tampilan Jarak (m / km) let distanceDisplay = ""; if (distanceInMeters < 1000) { distanceDisplay = `${Math.round(distanceInMeters)} meter`; } else { distanceDisplay = `${(distanceInMeters / 1000).toFixed(2)} km`; } document.getElementById('fareText').innerText = `Rp ${computedFare.toLocaleString('id-ID')} (${distanceDisplay})`; } }
